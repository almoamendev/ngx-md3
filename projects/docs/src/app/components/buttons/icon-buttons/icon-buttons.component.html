Floating Action Buttons

  • @Input() button-size: 'x-small' | 'small' | 'medium' | 'large' | 'x-large' = 'small';
  • @Input() button-shape: 'round' | 'square' = 'round';
  • @Input() button-type: 'filled' | 'tonal' | 'outlined' | 'standard' = 'filled';
  • @Input() button-width: 'default' | 'narrow' | 'wide' = 'default';
